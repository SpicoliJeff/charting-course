/************************************************************** CAGED OVERLAYS — Highlights fretboard regions corresponding to the 5 classic CAGED shapes. This version keeps it simple: each shape is defined as a fret range per string. **************************************************************/ const CAGED_SHAPES = { C: { start: 0, end: 4 }, A: { start: 3, end: 7 }, G: { start: 5, end: 9 }, E: { start: 7, end: 11 }, D: { start: 10, end: 14 } }; /************************************************************** APPLY CAGED — Adds a subtle background tint to frets that fall inside the active CAGED shape’s fret range. **************************************************************/ function applyCAGED() { if (!state.showCAGED) return; const fb = document.getElementById("fretboard"); const cells = fb.children; // Pick a shape based on the root note index (cyclic) const shapeNames = Object.keys(CAGED_SHAPES); const shape = CAGED_SHAPES[ shapeNames[ noteIndex(state.root) % 5 ] ]; for (let i = 0; i < cells.length; i++) { const fret = i % 25; if (fret >= shape.start && fret <= shape.end) { cells[i].style.background = "#334155"; } } } /************************************************************** SCALE‑RUN NODES — Defines simple ascending/descending scale‑run patterns. These are fret‑position templates that get highlighted when "Runs" is toggled. **************************************************************/ const RUN_PATTERNS = [ // 3‑note-per‑string style pattern (generic major scale layout) { frets: [0, 2, 4] }, { frets: [0, 2, 4] }, { frets: [1, 3, 5] }, { frets: [1, 3, 5] }, { frets: [2, 3, 5] }, { frets: [2, 3, 5] } ]; /************************************************************** APPLY SCALE‑RUNS — Highlights specific frets on each string based on the RUN_PATTERNS array. This is a visual guide for practicing scale‑run mechanics across the neck. **************************************************************/ function applyRuns() { if (!state.showRuns) return; const fb = document.getElementById("fretboard"); const cells = fb.children; for (let st = 0; st < RUN_PATTERNS.length; st++) { const pattern = RUN_PATTERNS[st].frets; for (let f = 0; f < pattern.length; f++) { const fret = pattern[f]; // Convert (string, fret) → cell index const index = st * 25 + fret; if (cells[index]) { cells[index].style.border = "2px solid #38bdf8"; // cyan highlight } } } } /************************************************************** POST‑PROCESS OVERLAYS — After building the fretboard grid, this function applies: - CAGED overlays - Scale‑run overlays It ensures overlays appear *on top* of the base fretboard. **************************************************************/ function applyOverlays() { applyCAGED(); applyRuns(); } /************************************************************** REBUILD FRETBOARD — Wrapper that rebuilds the fretboard grid and then applies overlays (CAGED + Runs). This ensures that overlays always appear on top of the base rendering. **************************************************************/ function rebuildFretboard() { buildFretboard(); applyOverlays(); } /************************************************************** OVERRIDE DEFAULT BUILD — Replace direct calls to buildFretboard() with rebuildFretboard() so overlays are always applied. **************************************************************/ function overrideBuildCalls() { // Replace the global buildFretboard reference with the wrapper const originalBuild = buildFretboard; buildFretboard = function() { originalBuild(); applyOverlays(); }; } // Activate override overrideBuildCalls(); /************************************************************** FINAL INTEGRATION — Ensures that after initialization, overlays (CAGED + Runs) are applied immediately so the user sees the correct visual state on first load. **************************************************************/ window.addEventListener("load", () => { // Rebuild with overlays applied rebuildFretboard(); });
